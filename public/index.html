export default async function handler(req, res) { let target = req.query.url; if (!target) return res.status(400).send('Search item missing'); // FIX: Correctly formats the secure, text-only search path for DuckDuckGo if (!target.includes('.')) { target = 'https://duckduckgo.com' + encodeURIComponent(target); } else if (!/^https?:\/\//i.test(target)) { target = 'https://' + target; } try { const response = await fetch(target, { method: 'GET', headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36', 'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8', 'Accept-Language': 'en-US,en;q=0.5' } }); if (!response.ok) { return res.status(response.status).send(`Website blocked the proxy connection. Error Code: ${response.status}`); } let html = await response.text(); // Security Patch: Injects code so links clicked inside the proxied site don't escape your proxy window const baseDomain = new URL(target).origin; html = html.replace(/href="\/([^"]*)"/g, `href="/api/proxy?url=${encodeURIComponent(baseDomain)}/$1"`); html = html.replace(/href="http([^"]*)"/g, (match, p1) => `href="/api/proxy?url=${encodeURIComponent('http' + p1)}"`); res.setHeader('Content-Type', 'text/html; charset=utf-8'); return res.status(200).send(html); } catch (err) { return res.status(500).send(`Network Connection Fault: ${err.message}`); } }
